# ==========================================
# Bài tập 3: Tăng cường bảo mật SSH (SSH Key + Chặn root login)
# ==========================================

# --- BƯỚC 1: Chạy trên máy cá nhân (Windows PowerShell) ---
# 1.1 Tạo cặp SSH Key Ed25519 (Private & Public Key)
ssh-keygen -t ed25519 -C "deploy@quickbite-vps" -f "$env:USERPROFILE\.ssh\id_ed25519_vps"

# 1.2 Xem nội dung Public Key (.pub) để copy lên VPS (TUYỆT ĐỐI KHÔNG chia sẻ file không có đuôi .pub)
Get-Content "$env:USERPROFILE\.ssh\id_ed25519_vps.pub"

# (Hoặc đẩy trực tiếp Public Key lên user 'deploy' trên VPS bằng 1 dòng PowerShell)
Get-Content "$env:USERPROFILE\.ssh\id_ed25519_vps.pub" | ssh root@<VPS_PUBLIC_IP> "mkdir -p /home/deploy/.ssh && cat >> /home/deploy/.ssh/authorized_keys && chmod 700 /home/deploy/.ssh && chmod 600 /home/deploy/.ssh/authorized_keys && chown -R deploy:deploy /home/deploy/.ssh"

# 1.3 BẮT BUỘC kiểm tra đăng nhập bằng SSH Key cho 'deploy' TRƯỚC KHI khóa mật khẩu
ssh -i "$env:USERPROFILE\.ssh\id_ed25519_vps" deploy@<VPS_PUBLIC_IP>


# --- BƯỚC 2: Chạy trên VPS (dưới quyền user 'deploy' có sudo hoặc 'root') ---
# 2.1 Cấu hình /etc/ssh/sshd_config và drop-in config (tránh bị cloud-init ghi đè)
sudo mkdir -p /etc/ssh/sshd_config.d
sudo tee /etc/ssh/sshd_config.d/99-hardening.conf > /dev/null <<'EOF'
PermitRootLogin no
PasswordAuthentication no
PubkeyAuthentication yes
KbdInteractiveAuthentication no
EOF

sudo sed -i 's/^#\?PermitRootLogin.*/PermitRootLogin no/' /etc/ssh/sshd_config
sudo sed -i 's/^#\?PasswordAuthentication.*/PasswordAuthentication no/' /etc/ssh/sshd_config

# 2.2 Kiểm tra cú pháp cấu hình SSH và khởi động lại dịch vụ SSH
sudo sshd -t
sudo systemctl restart ssh || sudo systemctl restart sshd


# --- BƯỚC 3: Kiểm chứng trên máy cá nhân (PowerShell) ---
# 3.1 Thử đăng nhập lại bằng root -> KẾT QUẢ MONG ĐỢI: Permission denied (publickey)
ssh root@<VPS_PUBLIC_IP>

# 3.2 Đăng nhập bằng user deploy dùng Private Key -> Thành công
ssh -i "$env:USERPROFILE\.ssh\id_ed25519_vps" deploy@<VPS_PUBLIC_IP>
